\begin{align*}
    J_p \ddot{p} &= \sum \tau_p \\
                 &= l_p F_b - l_p F_{g,b} - l_p F_f + l_p F_{g,f} \\
                 &= l_p (F_b - F_f + m_p g - m_p g) \\
                 &= l_p K_f (V_b - V_f) \\
                 &= L_1 V_d
\end{align*}

\begin{align*}
    J_e \ddot{e} &= \sum \tau_e \\
                 &= l_h \cos(e) (F_{g,f} + F_{g,b}) + l_h \cos(p) (F_f + F_b) + l_c \cos(e) F_{g,c} \\
                 &= (l_h (F_{g,f} + F_{g,b}) + l_c F_{g,c}) \cos(e) + l_h (F_f + F_b) \cos(p) \\
                 &= L_2 \cos(e) + l_h K_f (V_b + V_f) \cos(p) \\
                 &= L_2 \cos(e) + L_3 V_s \cos(p)
\end{align*}

\begin{align*}
    J_\lambda \ddot{\lambda} &= \sum \tau_\lambda \\
                 &= l_h \cos(e) \sin(p) (F_{g,f} + F_{g,b})\\
                 &= l_h \cos(e) \sin(p) K_f(V_b + V_f)\\
                 &= l_h \cos(e) \sin(p) K_f(V_b + V_f)\\
                 &= L_4 V_s \cos(e) \sin(p)\\
\end{align*}
