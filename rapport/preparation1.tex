\section{Preparation 1}
\subsection{Problem 1 - Equations of motion}
\begin{align*}
    J_p \ddot{p} &= \sum \tau_p \\
                 &= l_p F_b - l_p F_{g,b} - l_p F_f + l_p F_{g,f} \\
                 &= l_p (F_b - F_f + m_p g - m_p g) \\
                 &= l_p K_f (V_b - V_f) \\
                 &= L_1 V_d
\end{align*}

\begin{align*}
    J_e \ddot{e} &= \sum \tau_e \\
                 &= l_h \cos(e) (F_{g,f} + F_{g,b}) + l_h \cos(p) (F_f + F_b) + l_c \cos(e) F_{g,c} \\
                 &= (l_h (F_{g,f} + F_{g,b}) + l_c F_{g,c}) \cos(e) + l_h (F_f + F_b) \cos(p) \\
                 &= L_2 \cos(e) + l_h K_f (V_b + V_f) \cos(p) \\
                 &= L_2 \cos(e) + L_3 V_s \cos(p)
\end{align*}

\begin{align*}
    J_\lambda \ddot{\lambda} &= \sum \tau_\lambda \\
                 &= l_h \cos(e) \sin(p) (F_{g,f} + F_{g,b})\\
                 &= l_h \cos(e) \sin(p) K_f(V_b + V_f)\\
                 &= L_4 V_s \cos(e) \sin(p)\\
\end{align*}
\subsection{Problem 2 - Linearization}

\begin{align*}
    J_p \ddot{p} &= L_1 V_d \\
    J_p &=2 m_p l_p^2 \\
    2 m_p l_p^2 \ddot{p} &=L_1 V_d \\
    \ddot{p} &=K_1 V_d \\
\end{align*}

\begin{align*}
    J_e \ddot{e} &= L_2 \cos(e) + L_3 V_s \cos(p) \\
    J_e \ddot{e} &= L_2 + L_3 V_s  \\
    \ddot{e} &= \frac{L_2}{J_e} + \frac{L_3}{J_e} V_s  \\
    \ddot{e} &= \frac{L_3}{J_e} (V_s +\frac{L_2}{L_3})  \\
    K_2 &=\frac{L_3}{J_e} \\
    J_e &= m_c l_c^2 + 2m_p(l_h^2+l_p^2)\\
    \tilde{V_s}&=(V_s - V_{s,0}) \\
    V_{s,0} &=-\frac{L_2}{L_3} \\
    \ddot{e} &= K_2 \tilde{V_s}  \\
\end{align*}

\begin{align*}
    J_\lambda \ddot{e} &= L_4 V_s \cos(e) \sin(p)\\
    \ddot{\lambda} &= \frac{L_4}{J_\lambda} V_s p\\
    \ddot{\lambda} &= K_3 V_s p\\
\end{align*}

\begin{align*}
    J_\lambda \ddot{\lambda}&= L_4 V_s \cos(e) \sin(p)\\
    J_\lambda \ddot{\lambda}&= -L_4 V_s p \\
    V_s& \simeq V_{s,0} \\
    \ddot{\lambda}&= K_3 p \\
\end{align*}

Uenig, $V_s$ er en variabel, mens $V_{s,0}$ er en konstant. Man kan justere vinkelakselerasjonen ved å justere $V$ opp og ned. Jeg mener det riktige blir:

\begin{align*}
    J_\lambda \ddot{\lambda}&= L_4 V_s \cos(e) \sin(p)\\
    J_\lambda \ddot{\lambda}&= \frac{L_4}{J_\lambda} V_s p \\
    \ddot{\lambda}&= K_3 V_s p \\
\end{align*}

\subsection{Problem 3 - PD control}

\begin{align*}
    V_d &= K_{pp} (p_c - p) - K_{pd} \dot{p}\\
    \ddot{p} &= K_1 (K_{pp} (p_c - p) - K_{pd} \dot{p})\\
    S^2 p &= K_1 K_{pp} p_c - K_1 K_{pp} p - K_1 K_{pd} S p\\
    (S^2 + K_1 K_{pd} S + K_1 K_{pp}) p &= K_1 K_{pp} p_c\\
    \frac{p}{p_c} &= \frac{K_1 K_{pp}}{S^2 + K_1 K_{pd} S + K_1 K_{pp}}\\
    G(s) &= \frac{K_1 K_{pp}}{S^2 + K_1 K_{pd} S + K_1 K_{pp}}\\
\end{align*}

\subsection{Problem 4 - Pole Placement}

\begin{align*}
    (s - \lambda_1) (s - \lambda_2) &= s^2 - (\lambda_1 + \lambda_2) s + \lambda_1 \lambda_2\\
    s^2 - (\lambda_1 + \lambda_2) s + \lambda_1 \lambda_2 &= s^2 + K_1 K_{pd} s + K_1 K_{pp}\\
    K_1 K_{pp} &= \lambda_1 \lambda_2\\
    K_{pp} &= \frac{\lambda_1 \lambda_2}{K_1}\\
    K_1 K_{pd} &= -(\lambda_1 + \lambda_2)\\
    K_{pd} &= \frac{-(\lambda_1 + \lambda_2)}{K_1}\\
\end{align*}
